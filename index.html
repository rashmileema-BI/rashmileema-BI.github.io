-- Top 3 Worst Delayed Product Categories per Market (CTEs & Window Functions)
WITH category_delays AS (
  SELECT 
    market,
    category_name,
    COUNT(*) AS total_orders,
    ROUND(AVG(shipping_delay_days), 2) AS avg_delay_days,
    ROUND(AVG(late_delivery_risk) * 100.0, 2) AS late_risk_pct,
    ROW_NUMBER() OVER (
      PARTITION BY market 
      ORDER BY AVG(late_delivery_risk) DESC
    ) AS rank_in_market
  FROM orders
  GROUP BY market, category_name
  HAVING COUNT(*) >= 100
)
SELECT market, category_name, total_orders, avg_delay_days, late_risk_pct
FROM category_delays
WHERE rank_in_market <= 3
ORDER BY market, rank_in_market;
