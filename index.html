vw_player_risk_scores.sql · Fraud Hub 
-- Sum weighted rule hits per player, then band the score
WITH f AS (
  SELECT player_id, rule_code, MIN(detail) AS detail
  FROM core.risk_flags
  GROUP BY player_id, rule_code
)
SELECT f.player_id, p.country,
       SUM(w.points) AS risk_score,
       CASE WHEN SUM(w.points) >= 60 THEN 'High'
            WHEN SUM(w.points) >= 40 THEN 'Medium'
            ELSE 'Low' END AS risk_band,
       STRING_AGG(f.rule_code, '; ') WITHIN GROUP (ORDER BY f.rule_code) AS rules
FROM f
JOIN core.rule_weights AS w ON w.rule_code = f.rule_code
JOIN core.players AS p ON p.player_id = f.player_id
GROUP BY f.player_id, p.country;
